> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bring PatataTube's existing over-the-air iOS release (`./deploy`) up to the full spec in `~/c/agents-configs/docs/ios-ota-rollout/pipeline-prompt.md`, using `~/c/vvterm` as the reference, without breaking what already works.

**Architecture:** `./deploy` (Ruby, repo root) stays the single entry point. Its helpers move under `scripts/deploy/`: `release.rb` (pure CLI, version and notes helpers), `install_page.rb` (HTML), `ipa_builder.rb` (moved from `ios/ipa_builder.rb`: xcodebuild, signing, codesign probe) and `uploader.rb` (atomic copy into file_server's directory). Signing identity moves out of `project.yml` into a committed `ios/PatataTube/Config/PatataTube.xcconfig`, which a gitignored `Config/Local.xcconfig` overrides. The app gets an Info.plist-driven "Install Latest" dynamic quick action. DevLog gets an Info.plist URL that points it at a new stdlib server, `scripts/devlog/server.py`, on port 19485.

**Tech Stack:** Ruby 3.2 (stdlib + minitest), XcodeGen, xcodebuild (Xcode 15.3+ `release-testing` export), Swift 6 / SwiftUI / UIKit scene delegate, swift-testing + XCTest, Python 3 stdlib (`http.server`) + pytest.

```ruby
# scripts/deploy/release.rb
# frozen_string_literal: true

# Pure helpers for ../../deploy: nothing here touches git, Xcode or the network,
# so all of it is unit-tested (scripts/deploy/test/release_test.rb).

module Release
  class UsageError < StandardError; end

  TAG_PREFIX = "patatatube-ota-v"

  # Merge/release/chore noise that says nothing about what changed on the device.
  # "Release iOS v" is the pre-2.7.29 release commit title; keep filtering it.
  NOTE_NOISE = /\A(Release (iOS v|PatataTube OTA )|Merge |Commit pending changes|Bump |wip\b)/i
  MAX_NOTES = 8

  module_function

  # Minimal dotenv: KEY=VALUE lines, "#" comments, optional surrounding quotes.
  # Never overrides a variable already exported in the shell, so a one-off
  # `PATATATUBE_PUBLISH_DIR=/tmp/x ./deploy` beats the file. The .env is shared
  # with the Python backend; this only reads the keys it is asked for later.
  def load_dotenv(path)
    return unless File.exist?(path)

    File.readlines(path).each do |line|
      line = line.strip
      next if line.empty? || line.start_with?("#")

      key, value = line.split("=", 2)
      next unless key && value && key.match?(/\A[A-Za-z_][A-Za-z0-9_]*\z/)

      ENV[key] ||= value.strip.gsub(/\A["']|["']\z/, "")
    end
  end

  # Repeatable "--flag value" / "--flag=value" options, removed from argv so the
  # version argument stays positional wherever it sits.
  def take_option(argv, *names)
    values = []
    i = 0
    while i < argv.length
      arg = argv[i]
      if names.include?(arg)
        value = argv[i + 1]
        raise UsageError, "#{arg} needs a value" if value.nil? || value.start_with?("--")

        values << value
        argv.slice!(i, 2)
      elsif (eq = arg.match(/\A(--[a-z-]+)=(.*)\z/m)) && names.include?(eq[1])
        values << eq[2]
        argv.slice!(i, 1)
      else
        i += 1
      end
    end
    values
  end

  def clean_notes(raw)
    raw.flat_map { |n| n.split("\n") }
       .map { |n| n.strip.sub(/\A[-*•]\s*/, "") }
       .reject(&:empty?)
  end

  def bump(current, arg)
    return arg if arg.to_s.match?(/\A\d+\.\d+\.\d+\z/)

    major, minor, patch = current.split(".").map(&:to_i)
    case arg
    when "major" then "#{major + 1}.0.0"
    when "minor" then "#{major}.#{minor + 1}.0"
    when "patch", nil then "#{major}.#{minor}.#{patch.to_i + 1}"
    else raise UsageError, "bad argument '#{arg}' (use patch|minor|major or an X.Y.Z version)"
    end
  end

  def tag_for(version) = "#{TAG_PREFIX}#{version}"

  # Releases up to 2.7.28 were tagged plain vX.Y.Z. Still checked so an old
  # version number can never be released a second time.
  def legacy_tag_for(version) = "v#{version}"

  def filter_subjects(subjects)
    subjects.map(&:strip).reject { |s| s.empty? || s.match?(NOTE_NOISE) }.uniq.first(MAX_NOTES)
  end

  def release_notes(version:, summary:, notes:, subjects:, devlog:)
    bullets = notes.empty? ? filter_subjects(subjects) : notes
    headline = summary || (bullets.empty? ? "Release #{version}." : "What's new in #{version}:")
    headline = "[DEVLOG] #{headline}" if devlog
    lines = [headline]
    lines << "" unless bullets.empty?
    lines.concat(bullets.map { |b| "• #{b}" })
    if devlog
      lines << ""
      lines << "Diagnostics build: sends DevLog records to the Mac's devlog server."
    end
    lines.join("\n")
  end

  # project.yml (XcodeGen) is the version's source of truth here; the
  # .xcodeproj is generated and gitignored, so the bump goes into the YAML.
  def bump_project_yml(yml, version:, build:)
    unless yml.match?(/^\s*MARKETING_VERSION:/) && yml.match?(/^\s*CURRENT_PROJECT_VERSION:/)
      raise UsageError, "project.yml needs both MARKETING_VERSION and CURRENT_PROJECT_VERSION"
    end

    yml.sub(/^(\s*MARKETING_VERSION:\s*)"?[\d.]+"?/, "\\1\"#{version}\"")
       .sub(/^(\s*CURRENT_PROJECT_VERSION:\s*)"?\d+"?/, "\\1\"#{build}\"")
  end
end
```

```ruby
# frozen_string_literal: true

# ipa_builder.rb
#
# Builds a PatataTube .ipa from source, together with the OTA manifest.plist an
# `itms-services://` link points at. Used by ../../deploy and ../../ios/refresh-ipa.rb.
#
# The .ipa is signed **ad hoc** with the paid Apple Developer Program team from
# ios/PatataTube/Config/Local.xcconfig (gitignored; it overrides the committed
# defaults in Config/PatataTube.xcconfig). That is what lets iOS install it
# straight from Safari: no AltStore, no cable, and a signature good for a year.
# The price is that an Ad Hoc profile only covers **devices registered on the
# portal when the build is made**: a device added later needs a fresh ./deploy.
#
# PATATATUBE_UNSIGNED=1 falls back to an unsigned Payload/ zip, so a broken
# signing setup never blocks getting a binary. Safari cannot install that one.

require "fileutils"
require "shellwords"
require "tmpdir"

module IpaBuilder
  APP_NAME    = "PatataTube"
  SCHEME      = "PatataTube"
  REPO_ROOT   = File.expand_path("../..", __dir__)
  PROJECT_DIR = File.join(REPO_ROOT, "ios", APP_NAME)   # ios/PatataTube
  PROJECT     = "#{APP_NAME}.xcodeproj"
  PROJECT_YML = File.join(PROJECT_DIR, "project.yml")
  INFO_PLIST  = File.join(PROJECT_DIR, "Sources", "Info.plist")
  LOCAL_XCCONFIG = File.join(PROJECT_DIR, "Config", "Local.xcconfig")
  XCCONFIGS = [LOCAL_XCCONFIG, File.join(PROJECT_DIR, "Config", "PatataTube.xcconfig")].freeze

  module_function

  def bold(str) = "\e[1m#{str}\e[0m"
  def green(str) = "\e[32m#{str}\e[0m"
  def red(str) = "\e[31m#{str}\e[0m"
  def step(msg) = puts "\n#{bold("==> #{msg}")}"

  def die(msg)
    warn red("error: #{msg}")
    exit 1
  end

  def run(cmd, chdir:, hint: nil)
    puts "    $ #{cmd}"
    return if system(cmd, chdir: chdir)

    die(["command failed: #{cmd}", hint].compact.join("\n"))
  end

  def unsigned? = ENV["PATATATUBE_UNSIGNED"] == "1"

  # A build setting as Xcode resolves it: Local.xcconfig wins over the committed
  # PatataTube.xcconfig. Empty values fall through to the next file.
  def xcconfig_value(key, paths = XCCONFIGS)
    paths.each do |path|
      next unless File.exist?(path)

      value = File.read(path)[/^\s*#{Regexp.escape(key)}\s*=[ \t]*(\S*)/, 1]
      return value if value && !value.empty?
    end
    nil
  end

  def bundle_id = xcconfig_value("PATATATUBE_BUNDLE_ID")

  # PATATATUBE_TEAM_ID overrides Local.xcconfig for a one-off.
  def team_id
    id = ENV["PATATATUBE_TEAM_ID"] || xcconfig_value("DEVELOPMENT_TEAM")
    die("no DEVELOPMENT_TEAM in #{LOCAL_XCCONFIG} and no PATATATUBE_TEAM_ID set (see ios/install.md)") unless id
    id
  end

  # Preflight for a signed build: the committed defaults build in a fresh clone,
  # but cannot be signed for distribution.
  def check_signing_config!
    return if unsigned?

    team_id
    id = bundle_id
    return unless id.nil? || id.start_with?("com.example.")

    die("PATATATUBE_BUNDLE_ID is #{id.inspect}. Create #{LOCAL_XCCONFIG} with your registered " \
        "App ID and team (see ios/install.md), or set PATATATUBE_UNSIGNED=1 for an unsigned build.")
  end

  def signing_hint
    <<~TXT.chomp
      Signing failed. Check:
        - Xcode -> Settings -> Accounts lists the Apple ID for team #{ENV["PATATATUBE_TEAM_ID"] || xcconfig_value("DEVELOPMENT_TEAM")}.
        - The device is registered on developer.apple.com (Devices) BEFORE this build:
          an ad hoc profile only covers devices registered when it is made.
        - The App ID #{bundle_id} exists under Identifiers: automatic signing creates
          certificates and profiles, but never App IDs.
      To get a binary anyway: PATATATUBE_UNSIGNED=1 ./deploy
    TXT
  end

  # xcodebuild needs a full Xcode, not the Command Line Tools. Honour an existing
  # DEVELOPER_DIR, otherwise point at the newest /Applications/Xcode*.app.
  def resolve_developer_dir
    env = ENV["DEVELOPER_DIR"]
    return env if env && Dir.exist?(env)

    xcode = Dir.glob("/Applications/Xcode*.app").max
    die("no Xcode.app found; install Xcode or set DEVELOPER_DIR") unless xcode
    File.join(xcode, "Contents", "Developer")
  end

  def marketing_version
    m = File.read(PROJECT_YML)[/^\s*MARKETING_VERSION:\s*"?([\d.]+)"?/, 1]
    die("no MARKETING_VERSION in #{PROJECT_YML}") unless m
    m
  end

  def current_project_version
    m = File.read(PROJECT_YML)[/^\s*CURRENT_PROJECT_VERSION:\s*"?(\d+)"?/, 1]
    die("no CURRENT_PROJECT_VERSION in #{PROJECT_YML}") unless m
    m
  end

  # Over SSH, or after the Mac slept, the login keychain is locked and codesign
  # cannot use the signing key. The archive would then die at its first CodeSign
  # step (errSecInternalComponent) after minutes of compiling. Sign a throwaway
  # copy of /usr/bin/true as a probe; if that fails, unlock the keychain (prompts
  # for the login password) and probe again.
  def ensure_codesign_can_sign!
    return if unsigned?

    probe = File.join(Dir.tmpdir, "#{APP_NAME}-codesign-probe-#{Process.pid}")
    FileUtils.cp("/usr/bin/true", probe)
    sign = "codesign -f -s 'Apple Development' #{Shellwords.escape(probe)} >/dev/null 2>&1"
    return if system(sign)

    puts red("    codesign cannot use your signing key -- the login keychain is locked.")
    keychain = File.expand_path("~/Library/Keychains/login.keychain-db")
    system("security unlock-keychain #{Shellwords.escape(keychain)}") or
      die("could not unlock the login keychain")
    system(sign) or
      die("codesign still fails after unlocking; try: codesign -f -s 'Apple Development' /some/file")
  ensure
    FileUtils.rm_f(probe) if probe
  end

  def xml_escape(str)
    str.to_s.gsub("&", "&amp;").gsub("<", "&lt;").gsub(">", "&gt;")
  end

  # ExportOptions.plist for `xcodebuild -exportArchive`.
  #
  # `release-testing` is Xcode 15.3+'s name for ad hoc: it re-signs with the
  # team's distribution certificate against an Ad Hoc profile that embeds the
  # registered UDIDs. Automatic signing creates the certificate and profile, but
  # NOT the App ID, so PATATATUBE_BUNDLE_ID must already exist on the portal.
  #
  # The `manifest` dict makes xcodebuild write the manifest.plist an
  # itms-services:// link needs, so the manifest can never drift from the binary.
  # Its appURL is the .ipa's final public URL, known before the build.
  #
  # thinning `<none>` keeps one universal .ipa that a single URL can serve.
  def export_options_plist(path, manifest:)
    entries = [
      "<key>method</key><string>release-testing</string>",
      "<key>teamID</key><string>#{xml_escape(team_id)}</string>",
      "<key>signingStyle</key><string>automatic</string>",
      "<key>stripSwiftSymbols</key><true/>",
      "<key>thinning</key><string>&lt;none&gt;</string>",
      "<key>destination</key><string>export</string>",
    ]

    if manifest
      entries << <<~PLIST.chomp
        <key>manifest</key>
        <dict>
          <key>appURL</key><string>#{xml_escape(manifest[:app_url])}</string>
          <key>displayImageURL</key><string>#{xml_escape(manifest[:icon_url])}</string>
          <key>fullSizeImageURL</key><string>#{xml_escape(manifest[:icon_url])}</string>
        </dict>
      PLIST
    end

    File.write(path, <<~PLIST)
      <?xml version="1.0" encoding="UTF-8"?>
      <!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
      <plist version="1.0">
      <dict>
      #{entries.join("\n")}
      </dict>
      </plist>
    PLIST
    path
  end

  # Settings passed on the xcodebuild command line. The URLs go here, not in an
  # xcconfig, because xcconfig reads the `//` in a URL as a comment. The command
  # line is also the only place that reaches both the app target and the
  # PatataTubeKit SwiftPM package, which is why DEVLOG goes here too.
  def extra_build_settings(ota_manifest_url:, devlog_url:, instrumented:)
    s = +""
    s << "PATATATUBE_OTA_MANIFEST_URL=#{Shellwords.escape(ota_manifest_url)} " if ota_manifest_url
    s << "PATATATUBE_DEVLOG_URL=#{Shellwords.escape(devlog_url)} " if devlog_url
    s << "SWIFT_ACTIVE_COMPILATION_CONDITIONS='$(inherited) DEVLOG' " if instrumented
    s
  end

  # Builds the .ipa and returns its path, inside a fresh tmp dir the caller owns
  # and must remove. With `manifest:`, manifest.plist is written beside the .ipa.
  def build(manifest: nil, ota_manifest_url: nil, devlog_url: nil, instrumented: false)
    die("no xcodegen on PATH (brew install xcodegen)") if `which xcodegen`.empty?
    die("project dir not found: #{PROJECT_DIR}") unless Dir.exist?(PROJECT_DIR)

    ENV["DEVELOPER_DIR"] = resolve_developer_dir
    puts "    DEVELOPER_DIR=#{ENV['DEVELOPER_DIR']}"

    step "Regenerating Xcode project (xcodegen)"
    run("xcodegen generate", chdir: PROJECT_DIR)

    work    = Dir.mktmpdir("#{APP_NAME}-ipa-")
    archive = File.join(work, "#{APP_NAME}.xcarchive")

    signing = unsigned? ? "CODE_SIGNING_ALLOWED=NO " : "DEVELOPMENT_TEAM=#{team_id} "
    extra = extra_build_settings(ota_manifest_url: ota_manifest_url, devlog_url: devlog_url, instrumented: instrumented)

    step "Archiving (xcodebuild)#{instrumented ? ' [DEVLOG]' : ''}#{unsigned? ? ' [UNSIGNED]' : ''}"
    run(
      "xcodebuild " \
      "-project #{Shellwords.escape(PROJECT)} " \
      "-scheme #{Shellwords.escape(SCHEME)} " \
      "-configuration Release " \
      "-destination 'generic/platform=iOS' " \
      "-archivePath #{Shellwords.escape(archive)} " \
      "#{signing}#{extra}" \
      "-allowProvisioningUpdates " \
      "archive",
      chdir: PROJECT_DIR,
      hint: unsigned? ? nil : signing_hint
    )
    die("archive not produced: #{archive}") unless Dir.exist?(archive)
    puts green("    Built: #{archive}")

    return package_unsigned(archive, work) if unsigned?

    step "Exporting signed .ipa (ad hoc)"
    options = export_options_plist(File.join(work, "ExportOptions.plist"), manifest: manifest)
    run(
      "xcodebuild -exportArchive " \
      "-archivePath #{Shellwords.escape(archive)} " \
      "-exportPath #{Shellwords.escape(work)} " \
      "-exportOptionsPlist #{Shellwords.escape(options)} " \
      "-allowProvisioningUpdates",
      chdir: PROJECT_DIR,
      hint: signing_hint
    )

    ipa = File.join(work, "#{APP_NAME}.ipa")
    die("export produced no .ipa in #{work}.\n#{signing_hint}") unless File.exist?(ipa)
    ipa
  end

  # PATATATUBE_UNSIGNED=1: zip the .app into a Payload/ by hand.
  def package_unsigned(archive, work)
    app_path = File.join(archive, "Products", "Applications", "#{APP_NAME}.app")
    die("no #{APP_NAME}.app inside archive: #{app_path}") unless Dir.exist?(app_path)

    step "Packaging .ipa (unsigned)"
    payload = File.join(work, "Payload")
    FileUtils.mkdir_p(payload)
    FileUtils.cp_r(app_path, payload)

    ipa = File.join(work, "#{APP_NAME}.ipa")
    # ditto --keepParent keeps Payload/ at the archive root and preserves the
    # symlinks inside embedded frameworks (plain zip can mangle them).
    run("ditto -c -k --sequesterRsrc --keepParent #{Shellwords.escape(payload)} #{Shellwords.escape(ipa)}", chdir: work)
    ipa
  end
end
```

```ruby
# scripts/deploy/uploader.rb
# frozen_string_literal: true

# Publishes release artifacts into the files/ directory that file_server
# (files.chiq.me) serves from this machine.
#
# Publishing is a copy into that directory, not an HTTP upload: request bodies
# over ~12 MB stall in the Cloudflare tunnel and come back as 524s.
#
# Every publish is pinned. The artifact lands at exactly the name given and
# overwrites what was there, which is what keeps the manifest and install page
# URLs fixed, so one Home Screen bookmark keeps installing the newest release.

require "fileutils"
require_relative "ipa_builder"

module Uploader
  DEFAULT_PUBLISH_DIR = File.expand_path("../file_server/files", IpaBuilder::REPO_ROOT)

  module_function

  # PATATATUBE_UPLOAD_BASE comes from the repo's gitignored .env (loaded by
  # ../../deploy) or the shell.
  def base_url
    base = ENV["PATATATUBE_UPLOAD_BASE"].to_s
    IpaBuilder.die("set PATATATUBE_UPLOAD_BASE (in .env) to the public URL of your file server, e.g. https://files.chiq.me") if base.empty?
    base.sub(%r{/+\z}, "")
  end

  def url_for(name) = "#{base_url}/files/#{name}"

  def publish_dir = ENV.fetch("PATATATUBE_PUBLISH_DIR", DEFAULT_PUBLISH_DIR)

  def check_publish_dir!
    dir = publish_dir
    return dir if File.directory?(dir)

    IpaBuilder.die(
      "the file_server files/ directory was not found at #{dir}.\n" \
      "  Releases are published by writing into it, because the service serving\n" \
      "  #{base_url} runs on this machine. Set PATATATUBE_PUBLISH_DIR if it moved."
    )
  end

  # Publishes `path` under exactly `name` and returns its public URL. It copies
  # beside the target and renames, so a device downloading a fixed URL never
  # gets a half-written file.
  def upload(path, name:)
    IpaBuilder.die("file to publish not found: #{path}") unless File.file?(path)
    IpaBuilder.die("unsafe publish name: #{name.inspect}") unless name == File.basename(name)

    target = File.join(check_publish_dir!, name)
    staged = "#{target}.publishing-#{Process.pid}"
    begin
      FileUtils.cp(path, staged)
      File.rename(staged, target)
    ensure
      FileUtils.rm_f(staged)
    end
    url_for(name)
  end
end
```

```ruby
#!/usr/bin/env ruby
# frozen_string_literal: true

# deploy
#
# One-command over-the-air release of the PatataTube iOS app: no App Store, no
# TestFlight, no cable. (The rest of this repo is the Python server; ./serve
# runs that. This script only ships the iOS app.)
#
#   1. preflight: on main, clean tree, release tag unused, publish dir present,
#      signing config present, codesign able to sign
#   2. bump MARKETING_VERSION + CURRENT_PROJECT_VERSION in ios/PatataTube/project.yml
#      (XcodeGen's source of truth; the .xcodeproj is generated and gitignored)
#   3. build an ad-hoc-signed .ipa + its OTA manifest  (scripts/deploy/ipa_builder.rb)
#   4. publish .ipa, icon, manifest, install page      (scripts/deploy/uploader.rb)
#   5. commit "Release PatataTube OTA patatatube-ota-vX.Y.Z", tag it, push to `github`
#
# Usage:
#   ./deploy                   # bump patch (2.7.28 -> 2.7.29)
#   ./deploy patch|minor|major
#   ./deploy 2.8.0             # explicit version
#   ./deploy --yes             # skip the confirmation prompt
#   ./deploy --devlog          # diagnostics build: DevLog -> scripts/devlog/server.py
#                              # (--instrumented is the old name for the same flag)
#   ./deploy minor --summary "Faster offline downloads" \
#            --note "Download-all queues behind taps" --note "Fixed a stall"
#
# --summary is the headline and --note (repeatable) adds a bullet. With neither,
# the notes are the commit subjects since the last release tag, minus noise.
#
# Configuration:
#   .env at the repo root (gitignored, shared with the backend; never overrides
#   an exported variable):
#     PATATATUBE_UPLOAD_BASE=https://files.chiq.me   public URL of file_server (required)
#     PATATATUBE_PUBLISH_DIR=/path/to/files          the directory it serves
#                                                     (default ../file_server/files)
#   ios/PatataTube/Config/Local.xcconfig (gitignored):
#     PATATATUBE_BUNDLE_ID = com.patatatube.app
#     DEVELOPMENT_TEAM = <team id>
#   Environment:
#     PATATATUBE_UNSIGNED=1   unsigned build only: written to ios/build/, nothing
#                             is published, committed or tagged
#     PATATATUBE_TEAM_ID=...  one-off team override
#     PATATATUBE_DEVLOG_URL=  overrides --devlog's http://<LocalHostName>.local:19485/api/devlog
#     PATATATUBE_DEVLOG_PORT= DevLog port for --devlog's URL and scripts/devlog/server.py (default 19485)
#
# The manifest and install page have fixed URLs, so a Home Screen bookmark of
# https://files.chiq.me/files/patatatube-install.html installs the newest release
# forever. The app's "Install Latest" quick action opens the same manifest.

require_relative "scripts/deploy/release"
require_relative "scripts/deploy/ipa_builder"
require_relative "scripts/deploy/uploader"
require_relative "scripts/deploy/install_page"
require "fileutils"
require "net/http"
require "shellwords"

B = IpaBuilder
R = Release
REPO_ROOT = B::REPO_ROOT
REMOTE = "github"
BRANCH = "main"

R.load_dotenv(File.join(REPO_ROOT, ".env"))

PINNED_MANIFEST = "patatatube-manifest.plist"
PINNED_ICON     = "patatatube-icon.png"
PINNED_INSTALL  = "patatatube-install.html"
# The install page's name before 2.7.29. Devices have it bookmarked, so every
# release republishes the same page under it too.
LEGACY_INSTALL  = "3cf25ffc-754e-4a19-b4e1-bcc84cf02bd4-Install_PatataTube.html"
ICON_SOURCE     = File.join(B::PROJECT_DIR, "Sources/Assets.xcassets/AppIcon.appiconset/icon-1024.png")
MIN_OS          = File.read(B::PROJECT_YML)[/^\s*iOS:\s*"?([\d.]+)"?/, 1] || "18.0"

def git(args) = `git -C #{REPO_ROOT.shellescape} #{args}`

def usage!
  yield
rescue R::UsageError => e
  B.die(e.message)
end

# --- flags -------------------------------------------------------------------

ASSUME_YES = !ARGV.delete("--yes").nil?
DEVLOG = [ARGV.delete("--devlog"), ARGV.delete("--instrumented")].any? || ENV["PATATATUBE_DEVLOG"] == "1"
SUMMARY = usage! { R.take_option(ARGV, "--summary", "-s").last }
NOTES   = usage! { R.clean_notes(R.take_option(ARGV, "--note", "--notes", "-n")) }

# --- preflight (nothing here changes the working tree) -----------------------

branch = git("rev-parse --abbrev-ref HEAD").strip
B.die("on branch '#{branch}', not '#{BRANCH}'; releases are cut from #{BRANCH}") unless branch == BRANCH

dirty = git("status --porcelain")
B.die("working tree is dirty -- commit or stash before releasing:\n#{dirty}") unless dirty.strip.empty?

B.die("xcodegen not found (brew install xcodegen)") unless system("command -v xcodegen >/dev/null")
B.die("icon not found at #{ICON_SOURCE}") unless File.file?(ICON_SOURCE)

BASE = Uploader.base_url
MANIFEST_URL = Uploader.url_for(PINNED_MANIFEST)
ICON_URL     = Uploader.url_for(PINNED_ICON)
INSTALL_URL  = Uploader.url_for(PINNED_INSTALL)
ITMS_URL     = "itms-services://?action=download-manifest&url=#{MANIFEST_URL}"

current = B.marketing_version
version = usage! { R.bump(current, ARGV[0]) }
tag     = R.tag_for(version)
asset   = "#{B::APP_NAME}-#{version}.ipa"
download = Uploader.url_for(asset)

# The release tag is what guarantees a version is never released twice. The
# legacy vX.Y.Z tags (up to 2.7.28) count as taken too.
taken = [tag, R.legacy_tag_for(version)].find { |t| git("tag -l #{t.shellescape}").strip == t }
B.die("tag #{taken} already exists -- #{version} was already released; bump to a new version") if taken

publish_dir = Uploader.check_publish_dir!
B.die("#{asset} is already published in #{publish_dir} -- bump to a new version") if File.exist?(File.join(publish_dir, asset))

B.check_signing_config!
# Last, because it may prompt for the login password.
B.ensure_codesign_can_sign!

# --- release notes -----------------------------------------------------------

last_tag = git("describe --tags --abbrev=0 --match '#{R::TAG_PREFIX}*' --match 'v[0-9]*' 2>/dev/null").strip
range = last_tag.empty? ? "-n 20 HEAD" : "#{last_tag}..HEAD"
subjects = git("log --no-merges --pretty=%s #{range}").lines
DESCRIPTION = R.release_notes(version: version, summary: SUMMARY, notes: NOTES, subjects: subjects, devlog: DEVLOG)

build = B.current_project_version.to_i + 1

puts B.bold("\n==> Releasing #{B::APP_NAME} (OTA) #{current} -> #{version} (build #{build})#{B.unsigned? ? ' [UNSIGNED, build only]' : ''}")
puts B.bold("\n    What's New:")
DESCRIPTION.each_line { |l| puts "      #{l.chomp}" }
puts "\n    Install page: #{INSTALL_URL}"

if DEVLOG && !ASSUME_YES
  puts B.red(B.bold("\n  !  This publishes a DIAGNOSTICS build (DevLog)."))
  puts "     It sends runtime records to scripts/devlog/server.py. Ship a clean build over it afterwards."
  print "\n  Type 'yes' to continue: "
  B.die("aborted") unless $stdin.gets&.strip == "yes"
elsif !ASSUME_YES
  print "\n    Continue? [y/N] "
  B.die("aborted") unless $stdin.gets.to_s.strip.downcase.start_with?("y")
end

# --- write the bump; restore it on any exit before the commit ----------------
#
# From here on the tree is modified. A run that dies before the commit (a failed
# build, a failed publish, Ctrl-C, an exception) must put it back: a leftover
# bump would fail the NEXT run's clean-tree check, with an error pointing nowhere
# near this run. B.die calls exit, so at_exit covers every way out. Info.plist is
# included because the build runs xcodegen, which rewrites it.

MUTATED = [B::PROJECT_YML, B::INFO_PLIST].freeze
originals = MUTATED.to_h { |p| [p, File.read(p)] }
committed = false
at_exit { originals.each { |file, content| File.write(file, content) } unless committed }

File.write(B::PROJECT_YML, usage! { R.bump_project_yml(originals[B::PROJECT_YML], version: version, build: build) })

# --- build -------------------------------------------------------------------

devlog_url = nil
if DEVLOG
  override = ENV["PATATATUBE_DEVLOG_URL"].to_s
  # Each app on the macmini has its own DevLog port (vvterm 19480, patatatube 19485), so
  # the servers run side by side. PATATATUBE_DEVLOG_PORT moves both this URL and the server.
  port = ENV.fetch("PATATATUBE_DEVLOG_PORT", "19485")
  devlog_url = override.empty? ? "http://#{`scutil --get LocalHostName`.strip}.local:#{port}/api/devlog" : override
  puts "    DevLog: #{devlog_url}"
  begin
    res = Net::HTTP.get_response(URI("#{devlog_url}/health"))
    raise "HTTP #{res.code}" unless res.is_a?(Net::HTTPSuccess)
  rescue StandardError => e
    puts B.red("    devlog server not answering (#{e.message}); start it before installing: scripts/devlog/server.py")
  end
end

ipa = B.build(manifest: { app_url: download, icon_url: ICON_URL },
              ota_manifest_url: MANIFEST_URL, devlog_url: devlog_url, instrumented: DEVLOG)
work = File.dirname(ipa)

if B.unsigned?
  out_dir = File.join(REPO_ROOT, "ios", "build")
  FileUtils.mkdir_p(out_dir)
  out = File.join(out_dir, "#{B::APP_NAME}-#{version}-unsigned.ipa")
  FileUtils.mv(ipa, out)
  FileUtils.remove_entry(work)
  puts B.green("\n✓ Unsigned build: #{out} (#{(File.size(out) / 1_048_576.0).round(1)} MB)")
  puts "  Not published, committed or tagged; the version bump has been undone."
  puts "  Safari cannot install an unsigned .ipa; it needs a re-signing tool."
  exit 0
end

built_manifest = File.join(work, "manifest.plist")
B.die("export produced no manifest.plist -- cannot publish the OTA route") unless File.exist?(built_manifest)
puts B.green("    #{asset} — #{(File.size(ipa) / 1_048_576.0).round(1)} MB")

# --- publish -----------------------------------------------------------------
#
# Order matters: the manifest points at the .ipa and the icon, and the page
# points at the manifest, so each goes up only after what it references.

page = File.join(work, PINNED_INSTALL)
File.write(page, InstallPage.render(version: version, build: build, icon_url: ICON_URL,
                                    itms_url: ITMS_URL, notes: DESCRIPTION, min_os: MIN_OS))

B.step "Publishing to #{BASE} (#{publish_dir})"
puts "    #{Uploader.upload(ipa, name: asset)}"
puts "    #{Uploader.upload(ICON_SOURCE, name: PINNED_ICON)}"
puts "    #{Uploader.upload(built_manifest, name: PINNED_MANIFEST)}"
puts "    #{Uploader.upload(page, name: PINNED_INSTALL)}"
puts "    #{Uploader.upload(page, name: LEGACY_INSTALL)}"
FileUtils.remove_entry(work)

# --- commit, tag, push -------------------------------------------------------

B.step "Committing + pushing"
rel = ->(p) { p.delete_prefix("#{REPO_ROOT}/").shellescape }
B.run("git add #{rel.(B::PROJECT_YML)} #{rel.(B::INFO_PLIST)}", chdir: REPO_ROOT)
B.run("git commit -m #{"Release #{B::APP_NAME} OTA #{tag}".shellescape}", chdir: REPO_ROOT)
committed = true # the bump lives in a commit now; at_exit must not undo it
B.run("git tag #{tag.shellescape}", chdir: REPO_ROOT)

# git-sync's global post-commit hook pushes to the same remote the instant the
# commit lands, so our push can lose the race ("cannot lock ref ... is at X but
# expected Y") after the work is already done. On failure, ask the remote: if it
# holds our HEAD, that counts as success. Then make sure the tag got there too.
cmd = "git push #{REMOTE} #{BRANCH} refs/tags/#{tag}"
puts "    $ #{cmd}"
unless system(cmd, chdir: REPO_ROOT)
  local  = git("rev-parse HEAD").strip
  remote = git("ls-remote #{REMOTE} refs/heads/#{BRANCH}").split.first.to_s
  unless !local.empty? && local == remote
    B.die("command failed: #{cmd} (remote is at #{remote.empty? ? 'nothing' : remote[0, 7]}, we are at " \
          "#{local[0, 7]}). The release is already published; push by hand: git push #{REMOTE} #{BRANCH} #{tag}")
  end
  puts B.green("    remote already at #{local[0, 7]} — pushed by the git-sync hook; continuing")
  if git("ls-remote #{REMOTE} refs/tags/#{tag}").strip.empty?
    B.run("git push #{REMOTE} refs/tags/#{tag}", chdir: REPO_ROOT)
  end
end

puts B.green("\n✓ Published #{B::APP_NAME} #{version} (#{tag})")
puts "\n  Install / update on a device — open in Safari, tap Install:"
puts B.bold("    #{INSTALL_URL}")
puts "  Direct install link (for the ntfy tap target):"
puts "    #{ITMS_URL}"
if DEVLOG
  puts B.red(B.bold("\n  !  #{tag} is a DIAGNOSTICS build (DevLog)."))
  puts "     Records land in log/devlog.jsonl via scripts/devlog/server.py. When done:"
  puts B.bold("       ./deploy patch")
end
```

```swift
    static func reset() { lock.withLock { urls = [] } }

    override class func canInit(with request: URLRequest) -> Bool { true }
    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }
    override func startLoading() { Self.lock.withLock { Self.urls.append(request.url!) } }
    override func stopLoading() {}
```

```swift
import Foundation

/// Sends files (screenshots and the like) to scripts/devlog/server.py, which
/// stores them under log/screens/<name>. Files skip the record buffer, because
/// one image would crowd out many records. When too many are already in flight,
/// new ones are dropped, so the observed code never queues behind the network.
public final class DevLogUploader: @unchecked Sendable {
    public static let maxInFlight = 4
    /// Matches the server's per-file cap.
    public static let maxBytes = 8 * 1024 * 1024

    private let uploadURL: URL
    private let session: URLSession
    private let lock = NSLock()
    private var inFlight = 0

    public init(endpoint: URL, configuration: URLSessionConfiguration = DevLogHTTPSink.defaultConfiguration()) {
        self.uploadURL = endpoint.appendingPathComponent("upload")
        self.session = URLSession(configuration: configuration)
    }

    /// Returns false when the file was dropped (cap reached, too big, bad name).
    @discardableResult
    public func upload(name: String, data: Data) -> Bool {
        guard data.count <= Self.maxBytes,
              var components = URLComponents(url: uploadURL, resolvingAgainstBaseURL: false) else { return false }
        components.queryItems = [URLQueryItem(name: "name", value: name)]
        guard let url = components.url else { return false }

        let accepted: Bool = lock.withLock {
            guard inFlight < Self.maxInFlight else { return false }
            inFlight += 1
            return true
        }
        guard accepted else { return false }

        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/octet-stream", forHTTPHeaderField: "Content-Type")
        session.uploadTask(with: request, from: data) { [weak self] _, _, _ in
            guard let self else { return }
            self.lock.withLock { self.inFlight -= 1 }
        }.resume()
        return true
    }
}
```

```python
#!/usr/bin/env python3
"""Receives the DevLog records a `./deploy --devlog` build of PatataTube sends.

A phone cannot write to this Mac's filesystem, so the diagnostics build POSTs
batches of JSON records to /api/devlog. This server appends them, one per line,
to log/devlog.jsonl and echoes each to stdout, so a run can be watched live.
Files from DevLog.upload arrive at /api/devlog/upload?name=... and land in
log/screens/. Both are emptied on start, so they only hold the current run.

    scripts/devlog/server.py                        # 0.0.0.0:19485 -> log/devlog.jsonl
    PATATATUBE_DEVLOG_PORT=19500 scripts/devlog/server.py
    curl http://$(scutil --get LocalHostName).local:${PATATATUBE_DEVLOG_PORT:-19485}/api/devlog/health

Stdlib only. It listens on the LAN without auth, so it caps request, record and
file sizes instead of trusting the client. (Separate from the backend's
POST /api/devlog -> log/ios.jsonl, which Xcode device runs still use.)
"""

import json
import os
import re
import sys
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from threading import Lock
from urllib.parse import parse_qs, urlparse

PORT = int(os.getenv("PATATATUBE_DEVLOG_PORT", "19485"))
LOG_PATH = Path(os.getenv("PATATATUBE_DEVLOG_FILE", Path(__file__).resolve().parents[2] / "log" / "devlog.jsonl"))
UPLOADS_DIR = LOG_PATH.parent / "screens"
MAX_REQUEST_BYTES = 1024 * 1024
MAX_RECORDS_PER_REQUEST = 2000
# At the cap the file becomes devlog.jsonl.1, so at most two generations exist.
MAX_LOG_BYTES = 32 * 1024 * 1024
MAX_UPLOAD_BYTES = 8 * 1024 * 1024
MAX_UPLOADS_TOTAL_BYTES = 512 * 1024 * 1024
UPLOAD_NAME = re.compile(r"^[A-Za-z0-9._+-]{1,128}$")

write_lock = Lock()


def append(records):
    lines = []
    for record in records:
        try:
            lines.append(json.dumps(record, ensure_ascii=True, separators=(",", ":")))
        except (TypeError, ValueError):
            continue
    if not lines:
        return 0
    with write_lock:
        if LOG_PATH.exists() and LOG_PATH.stat().st_size >= MAX_LOG_BYTES:
            os.replace(LOG_PATH, LOG_PATH.with_suffix(LOG_PATH.suffix + ".1"))
        with LOG_PATH.open("a", encoding="utf-8") as handle:
            handle.write("\n".join(lines) + "\n")
    for line in lines:
        print(line, flush=True)
    return len(lines)


def uploads_size():
    return sum(p.stat().st_size for p in UPLOADS_DIR.iterdir() if p.is_file())


def save_upload(name, data):
    if not UPLOAD_NAME.match(name) or name.startswith("."):
        return False
    with write_lock:
        if uploads_size() + len(data) > MAX_UPLOADS_TOTAL_BYTES:
            return False
        (UPLOADS_DIR / name).write_bytes(data)
    return True


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path in ("/health", "/api/devlog/health"):
            self._reply(200, b"ok\n")
        else:
            self._reply(404, b"not found\n")

    def do_POST(self):
        url = urlparse(self.path)
        if url.path == "/api/devlog/upload":
            return self._upload(parse_qs(url.query).get("name", [""])[0])
        if url.path != "/api/devlog":
            return self._reply(404, b"not found\n")
        length = int(self.headers.get("Content-Length") or 0)
        if length <= 0 or length > MAX_REQUEST_BYTES:
            return self._reply(413, b"bad size\n")
        try:
            payload = json.loads(self.rfile.read(length))
            records = payload["records"]
            if not isinstance(records, list) or len(records) > MAX_RECORDS_PER_REQUEST:
                raise ValueError("records")
        except (ValueError, KeyError, TypeError):
            return self._reply(400, b"bad payload\n")
        append([r for r in records if isinstance(r, dict)])
        self._reply(204, b"")

    def _upload(self, name):
        length = int(self.headers.get("Content-Length") or 0)
        if length <= 0 or length > MAX_UPLOAD_BYTES:
            return self._reply(413, b"bad size\n")
        data = self.rfile.read(length)
        if not save_upload(name, data):
            return self._reply(400, b"bad name or uploads full\n")
        print(json.dumps({"kind": "upload", "name": name, "bytes": length}), flush=True)
        self._reply(204, b"")

    def _reply(self, status, body):
        self.send_response(status)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        if body:
            self.wfile.write(body)

    def log_message(self, *args):
        # Records already go to stdout; access lines would bury them.
        pass


def reset_logs():
    LOG_PATH.parent.mkdir(parents=True, exist_ok=True)
    LOG_PATH.write_text("")
    LOG_PATH.with_suffix(LOG_PATH.suffix + ".1").unlink(missing_ok=True)
    UPLOADS_DIR.mkdir(parents=True, exist_ok=True)
    for old in UPLOADS_DIR.iterdir():
        if old.is_file():
            old.unlink()


def make_server(port=PORT, host="0.0.0.0"):
    return ThreadingHTTPServer((host, port), Handler)


def main():
    reset_logs()
    server = make_server()
    print(f"devlog: listening on :{PORT}, writing {LOG_PATH}", file=sys.stderr, flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()
```
